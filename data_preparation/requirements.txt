# ===========================================================================
# Data-preparation pipeline dependencies.
# Tested with Python 3.10 + CUDA 12.1 on Linux.
#
# IMPORTANT: a plain `pip install -r requirements.txt` is NOT enough on its own.
# `cubvh` and `diso` are CUDA extensions compiled from source, so they need a
# build toolchain (nvcc 12.1 + a host gcc <= 12) AND torch installed *first*.
# Follow the ordered steps in the README ("Installation"); in short:
#
#   bash setup.sh                                   # Blender system libs (root)
#   conda create -n dataprep python=3.10 -y && conda activate dataprep
#   conda install -y -c "nvidia/label/cuda-12.1.0" cuda-toolkit
#   conda install -y -c conda-forge gcc_linux-64=12 gxx_linux-64=12 \
#                                   binutils_linux-64 sysroot_linux-64
#   export CUDA_HOME="$CONDA_PREFIX" NVCC_PREPEND_FLAGS="-ccbin $CXX" \
#          TORCH_CUDA_ARCH_LIST="8.0"               # 8.0 = A100; set to your GPU
#   pip install "setuptools<81" wheel ninja pybind11
#   pip install torch==2.4.0+cu121 torchvision==0.19.0+cu121 torchaudio==2.4.0+cu121
#   pip install -r requirements.txt --no-build-isolation
#
# Why the extra flags:
#   * `--no-build-isolation` lets cubvh/diso compile against the torch installed
#     above (their build step imports torch; isolated builds can't see it).
#   * `setuptools<81` is required: cubvh's setup.py imports `pkg_resources`,
#     which was removed in setuptools 81.
#
# For a byte-exact reproduction of the original environment use
# requirements-lock.txt instead (the same build prerequisites apply).
# ===========================================================================
